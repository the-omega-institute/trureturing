## Fixed reconstruction API

Copy the complete S19-S22 README Python APIs into external scratch as `s19_evidence.py` through `s22_evidence.py`. Copy the following block beside them as `s23_evidence.py`, use Python `-B`, and set `s23_evidence.ROOT` to the checkout root before calling it. Each process consumes one fixed snapshot; start a new process after edits. The cache belongs only to that snapshot. This extends the existing byte API and executes no archived code or mathematics. `verify()` checks all declared versions, aliases, whole-unit/structural coverage, addresses, source/unit inverses, parent preservation and report sections. Canonical closure and exhaustive original-schema span/reference checks are separate fixed representation evidence.

```python
"""Fixed S23 byte reconstruction, extending the S19-S22 evidence API."""
from pathlib import Path
from functools import lru_cache
from s19_evidence import strict_json, pointer, Evidence
from s20_evidence import identity, interval, S20Evidence
import s22_evidence as parent

ROOT = Path('.')
BASE = 'docs/reports/quantized-gh/s23-evidence/'

def file(path):
    p=Path(path)
    if p.is_absolute() or '..' in p.parts:
        raise ValueError('repository-relative physical path required')
    return (ROOT/p).read_bytes()

@lru_cache(None)
def read_json(path):
    return strict_json(file(path))

def catalog():
    return read_json(BASE+'composition.json')

def check(data,expected):
    e=dict(expected)
    for old,new in [('LF','lf'),('terminal_LF_bytes','terminal_lf'),('byte_count','bytes'),('lf_count','lf')]:
        if old in e:
            if new in e and e[new]!=e[old]:
                raise ValueError('conflicting original identity fields')
            e[new]=e[old]
    return parent.check(data,e)

def read_parts(name):
    entry=catalog()['versions'][name];d=read_json(entry['descriptor'])
    if d['schema']!='s23-evidence-entry-v1' or d['resource']!=name or d['original']!=entry['identity']:
        raise ValueError('wrong resource descriptor or identity')
    chunks=[];cursor=0;line=1
    for ordinal,part in enumerate(d['parts'],1):
        b=check(file(part['path']),part)
        if not b or part['ordinal']!=ordinal or part['lines']>750:
            raise ValueError('empty, unordered or oversized part')
        if part['byte_span']!=[cursor,cursor+len(b)] or part['line_span']!=[line,line+part['lines']-1]:
            raise ValueError('part byte/line coverage differs')
        if ordinal<len(d['parts']) and not b.endswith(b'\n'):
            raise ValueError('interior part must end on LF')
        chunks.append(b);cursor+=len(b);line+=part['lf']
    return check(b''.join(chunks),d['original'])

@lru_cache(None)
def resource(name):
    c=catalog();name=c['aliases'].get(name,name)
    if name.startswith(('s22:','s21:','s20:','s19:')):
        parent.ROOT=ROOT
        return parent.resource(name[4:] if name.startswith('s22:') else name)
    e=c['versions'][name];kind=e['kind']
    if kind=='parts':data=read_parts(name)
    elif kind=='file':data=file(e['path'])
    elif kind=='file_span':data=interval(file(e['path']),e['byte_span'])
    elif kind=='slices':data=b''.join(interval(resource(s['version']),s['byte_span']) for s in e['slices'])
    elif kind=='i44_inverse':data=recover()
    else:raise ValueError('unknown S23 representation kind')
    return check(data,e['identity'])

def at(name,json_pointer=''):
    return pointer(strict_json(resource(name)),json_pointer.removeprefix('#'))

def resolve(reference,owner='I51'):
    if owner=='I51':
        binding=catalog()['immutable_documents'][reference['document']]
        return pointer(strict_json(check(resource(binding['path']),binding)),reference['json_pointer'].removeprefix('#'))
    if owner=='I44':
        return at('I44',reference if isinstance(reference,str) else reference['json_pointer'])
    if owner.startswith('s22:'):
        parent.ROOT=ROOT
        return parent.resolve(reference,owner[4:])
    raise ValueError('explicit declared owner required')

def span(version,byte_span,expected=None):
    data=interval(resource(version),byte_span)
    return check(data,expected) if expected is not None else data

def recover():
    """Apply I44's original inverse to recover the complete I20 source."""
    m=at('I44');chunks=[];cursor=0
    for row in m['original_source_inverse']:
        target=row['inverse_read'];version='source-I44-original' if target['path']==catalog()['source_path'] else 'report-I44-original'
        b=span(version,[target['start_byte'],target['end_byte']],target)
        original=row['original']
        if [original['start_byte'],original['end_byte']]!=[cursor,cursor+len(b)]:
            raise ValueError('original I20 source partition differs')
        chunks.append(check(b,original));cursor+=len(b)
    return check(b''.join(chunks),m['original_I20_source'])

def section(anchor):
    nav=read_json(BASE+'navigation.json')
    row=next(s for s in nav['report_sections'] if anchor==s['anchor'] or anchor in s['aliases'])
    return span(nav['report_version'],row['byte_span'],row['identity'])

def verify():
    c=catalog()
    for name in c['versions']:resource(name)
    for alias,name in c['aliases'].items():
        if resource(alias)!=resource(name):raise ValueError('alias differs')
    source=resource('source-composed-I51')
    if source!=resource('source-S22-merged')+resource('tail-S23-current'):
        raise ValueError('source composition differs')
    if resource('source-S22-merged')!=resource('s22:source-composed-I50'):
        raise ValueError('original S22 version differs')
    m=at('I44')
    for field,version in c['map_versions']['I44'].items():check(resource(version),m[field])
    units=read_json(c['current_units'])['units']
    regions=[(u['span']['byte_start'],u['span']['byte_end_exclusive'],u['span']) for u in units]
    regions += [(r['byte_span'][0],r['byte_span'][1],r['identity']) for r in c['structural_regions']]
    cursor=0
    for start,end,expected in sorted(regions):
        if start!=cursor:raise ValueError('whole-unit/structural coverage differs')
        check(interval(source,[start,end]),expected);cursor=end
    if cursor!=len(source):raise ValueError('source suffix uncovered')
    for row in read_json(c['own_addresses'])['addresses']:
        original=resolve(row['binding_ref'])
        old=span(row['original_version'],row['original_span'],original['source'])
        cur=span(row['current_version'],row['current_span'],row['current_identity'])
        if row['unchanged'] and old!=cur:raise ValueError('unchanged whole unit differs')
    for p in read_json(c['unit_partitions'])['partitions']:
        old=resource(p['original_version']);cur=interval(source,p['current_span'])
        chunks=[];cursor=0
        for row in p['original_to_current_plus_archive']:
            data=cur if row['destination']=='current' else old
            b=check(interval(data,row['destination_span']),row['identity'])
            if row['original_span']!=[cursor,cursor+len(b)]:raise ValueError('unit inverse gap or overlap')
            chunks.append(b);cursor+=len(b)
        if b''.join(chunks)!=old:raise ValueError('unit inverse differs')
        chunks=[];cursor=0
        for row in p['current_from_original']:
            b=row['literal'].encode() if 'literal' in row else interval(old,row['original_span'])
            check(b,row['identity'])
            if row['current_span']!=[cursor,cursor+len(b)]:raise ValueError('current unit partition differs')
            chunks.append(b);cursor+=len(b)
        if b''.join(chunks)!=cur:raise ValueError('current unit reconstruction differs')
    inverse=read_json(c['source_inverse']);chunks=[];cursor=0
    for row in inverse['segments']:
        b=span(row['version'],row['byte_span'],row['identity'])
        if row['original_span']!=[cursor,cursor+len(b)]:raise ValueError('whole original source inverse differs')
        chunks.append(b);cursor+=len(b)
    if b''.join(chunks)!=resource('source-I44-original'):raise ValueError('whole original source differs')
    for row in read_json(c['navigation'])['report_sections']:section(row['anchor'])
    for row in read_json(BASE+'parent-contract-preservation.json')['parents']:
        check(resource(row['before_version']),row['before']);check(file(row['path']),row['current'])
    return dict(versions=len(c['versions']),aliases=len(c['aliases']),whole_units=len(units),structural_regions=len(c['structural_regions']),source=identity(source))
```
