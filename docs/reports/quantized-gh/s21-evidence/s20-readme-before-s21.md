```python
"""S20 fixed byte representation. Import Evidence from the public S19 recipe."""
from pathlib import Path
import json
import hashlib

BASE = 'docs/reports/quantized-gh/s20-evidence/'
SOURCE = 'docs/develop/theory/ARITHMETIC_BOUNDARY_QUANTIZATION.md'
REPORT = 'docs/reports/quantized-gh/balanced-prime-237-all-slabs-0910.md'

def identity(data):
    data.decode('utf-8', errors='strict')
    return dict(bytes=len(data), sha256=hashlib.sha256(data).hexdigest(),
                lf=data.count(b'\n'), terminal_lf=len(data)-len(data.rstrip(b'\n')),
                lines=data.count(b'\n')+int(bool(data) and not data.endswith(b'\n')))

def checked(data, expected):
    observed = identity(data)
    aliases = {'byte_count':'bytes', 'lf_count':'lf', 'terminal_lf_count':'terminal_lf'}
    for key, value in expected.items():
        key = aliases.get(key, key)
        if key in observed and (isinstance(value, bool) or value != observed[key]):
            raise ValueError('identity mismatch: '+key)
    return data

def interval(data, span):
    start, end = span
    if type(start) is not int or type(end) is not int or not 0 <= start <= end <= len(data):
        raise ValueError('invalid explicit byte interval')
    return data[start:end]

class S20Evidence:
    def __init__(self, root, s19):
        self.root, self.s19, self.cache = Path(root), s19, {}
        self.composition = self.read_json(BASE+'composition.json')

    def file(self, path):
        p = Path(path)
        if p.is_absolute() or '..' in p.parts:
            raise ValueError('repository-relative paths required')
        return (self.root/p).read_bytes()

    def read_json(self, path):
        # Use the same strict JSON parser as the fully read S19 recipe.
        from s19_evidence import strict_json
        return strict_json(self.file(path))

    def raw(self, name):
        name = self.composition.get('aliases', {}).get(name, name)
        if name in self.cache:
            return self.cache[name]
        if name.startswith('s19:'):
            return self.s19.raw(name[4:])
        entry = self.composition['versions'][name]
        kind = entry['kind']
        if kind == 'parts':
            descriptor = self.read_json(entry['descriptor'])
            if descriptor['schema'] != 's20-evidence-entry-v1' or descriptor['resource'] != name:
                raise ValueError('wrong historical resource descriptor')
            if descriptor['original'] != entry['identity']:
                raise ValueError('descriptor/version identity differs')
            chunks, cursor, line = [], 0, 1
            for ordinal, part in enumerate(descriptor['parts'], 1):
                b = checked(self.file(part['path']), part)
                if not b or part['lines'] > 800 or part['ordinal'] != ordinal:
                    raise ValueError('empty, oversized or unordered part')
                if part['byte_span'] != [cursor, cursor+len(b)]:
                    raise ValueError('part coverage gap or overlap')
                if part['line_span'] != [line, line+part['lines']-1]:
                    raise ValueError('part LF coverage differs')
                if ordinal < len(descriptor['parts']) and not b.endswith(b'\n'):
                    raise ValueError('interior part must end on LF')
                chunks.append(b)
                cursor += len(b)
                line += part['lf']
            data = checked(b''.join(chunks), descriptor['original'])
        elif kind == 'file':
            data = self.file(entry['path'])
        elif kind == 'file_span':
            data = interval(self.file(entry['path']), entry['byte_span'])
        elif kind == 'slices':
            data = b''.join(interval(self.raw(s['version']), s['byte_span']) for s in entry['slices'])
        elif kind == 'i29_recovery':
            old_map = self.json('I29')
            cursor, chunks = 0, []
            for row in old_map['original_byte_partition']:
                version = self.composition['i29_destination_owners'][row['destination_path']]
                b = checked(interval(self.raw(version), row['destination_byte_span']),
                            {'sha256':row['original_sha256']})
                if row['original_source_byte_span'] != [cursor, cursor+len(b)]:
                    raise ValueError('old source partition gap or overlap')
                chunks.append(b)
                cursor += len(b)
            data = b''.join(chunks)
        elif kind == 'remove':
            original = self.raw(entry['version'])
            start, end = entry['byte_span']
            if interval(original, [start,end]) != entry['literal'].encode('utf-8'):
                raise ValueError('historical inverse literal differs')
            data = original[:start]+original[end:]
        else:
            raise ValueError('unknown representation kind')
        self.cache[name] = checked(data, entry['identity'])
        return self.cache[name]

    def json(self, name):
        from s19_evidence import strict_json
        return strict_json(self.raw(name))

    def at(self, name, path=''):
        from s19_evidence import pointer
        return pointer(self.json(name), path)

    def section(self, anchor):
        nav = self.read_json(BASE+'navigation.json')
        section = next(s for s in nav['report_sections'] if s['anchor'] == anchor)
        return checked(interval(self.raw('report-S20-I29'), section['byte_span']), section['identity'])

    def verify(self):
        for name in self.composition['versions']:
            self.raw(name)
        source = self.raw('source-composed-I47')
        if source != self.raw('source-S19-merged')+self.raw('tail-S20-I29'):
            raise ValueError('source composition differs')
        units = self.read_json(BASE+'current-units.json')['units']
        structural = self.composition['structural_regions']
        regions = sorted([(u['span']['byte_start'], u['span']['byte_end_exclusive'], u['span']) for u in units]
                         +[(s['byte_span'][0], s['byte_span'][1], s['identity']) for s in structural])
        cursor = 0
        for start, end, expected in regions:
            if start != cursor:
                raise ValueError('whole-unit/structural coverage differs')
            checked(interval(source,[start,end]),expected)
            cursor = end
        if cursor != len(source):
            raise ValueError('source suffix uncovered')
        for row in self.read_json(BASE+'own-addresses.json')['addresses']:
            original = interval(self.raw('source-S20-pre-I29'),row['old_source_byte_span'])
            checked(original, {'sha256':row['old_sha256']})
            if row['current_unit'] is not None:
                u = units[row['current_unit_index']]
                checked(interval(source,[u['span']['byte_start'],u['span']['byte_end_exclusive']]),
                        {'sha256':row['current_unit']['sha256']})
        nav = self.read_json(BASE+'navigation.json')
        for section in nav['report_sections']:
            self.section(section['anchor'])
        return dict(versions=len(self.composition['versions']), whole_units=len(units),
                    structural_regions=len(structural), own_addresses=22,
                    report_sections=len(nav['report_sections']), source=identity(source))
```
